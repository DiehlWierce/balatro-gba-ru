# Gold Stake, seed GBARU1, Ante 3: buy the Eternal+Rental Joker ($1) and the Perishable one,
# try to sell the Eternal one (refused), win the next round: -$3 rent, Perishable counts down.

wait 460
press START
wait 60
press A
wait 60
poke32 {gv:stake} 7
pokestr {sym:s_seed_str} GBARU1

wait 10
press A
wait 120

# Small Blind, forced win
press A
wait 180

press A
press L
wait 1100
press A
wait 200

# Buy the first Joker (Eternal + Rental, $1)
press RIGHT
wait 20
press A
wait 60

# Back to the Shop row, buy the second Joker (Perishable, $4)
press DOWN
wait 20
press RIGHT
wait 20
press A
wait 60

# Cursor on owned Jokers: go to the first one and try to sell it (L): refused
press LEFT
wait 20
press L
wait 60

# Next round: back to the Shop row, Next round button
press DOWN
wait 20
press LEFT
wait 20
press UP
wait 20
press A
wait 200

# Big Blind, forced win: the rent appears in the rewards
press A

press L
wait 1500

wait 200
peek32 {gv:money}
