# Seed GBARU1: win the Small Blind (score forced), cash out, buy the first Shop Joker,
# leave the Shop and come back to the Blind Select screen.

# Splash screen, main menu, run setup
wait 460
press START
wait 60
press A
wait 60
shot 00_run_setup

# Fixed seed instead of typing it on the seed keyboard
pokestr {sym:s_seed_str} GBARU1
poke8 {sym:use_seed} 1
press DOWN
wait 10
press A
wait 120
shot 01_blind_select

# Select the Small Blind, play one card with the score forced above the requirement
press A
wait 180
shot 02_round
poke32 {gv:score} 100000
press A
press L
wait 1100
shot 03_round_end

# Cash out, show the description of the first Shop Joker and buy it
press A
wait 200
shot 04_shop
press RIGHT
wait 20
hold B
wait 90
shot 05_joker_description
release B
wait 30
press A
wait 60
shot 06_bought

# The cursor is now on the owned Jokers row: go back to the Shop and press "Next round"
press DOWN
wait 20
press LEFT
wait 20
press UP
wait 20
press A
wait 200
shot 07_next_blind_select
